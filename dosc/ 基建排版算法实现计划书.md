# 明日方舟基建最优排班算法 - 详细实现计划书

## 一、项目概述

### 1.1 目标

实现基于**贪心初始化 + 模拟退火优化**的基建排班算法，最大化生产效率，完整支持：
- 单干员基础效率
- 双人/多人联动技能
- 条件触发技能（如"每有N条赤金生产线"）
  - 全局变量技能（如"人间烟火"系统）


### 1.2 算法选择理由

| 算法 | 优点 | 缺点 | 适用性 |
|------|------|------|--------|
| 贪心 | 快速、简单 | 无法处理联动 | 仅作初始解 |
| 动态规划 | 最优解 | 状态空间爆炸 | 不适用 |
| ILP | 数学最优 | 联动需线性化，复杂 | 部分适用 |
| **模拟退火** | 灵活、可处理任意评估函数 | 非严格最优 | **最适合** |

### 1.3 预期效果

- 相比当前贪心算法，效率提升 **15-30%**
- 求解时间 **< 5秒**（410干员，18设施）
- 支持所有已知联动类型

### 4.2 评估流程

```cpp
double EfficiencyCalculator::evaluate(const InfrastructureState& state) {
    // 创建可修改的副本
    InfrastructureState eval_state = state;

    // ===== Phase 1: 计算全局变量 =====
    compute_global_variables(eval_state);
    compute_production_lines(eval_state);

    // ===== Phase 2: 计算各设施效率 =====
    double total = 0.0;

    for (const auto& [fac_id, operators] : eval_state.assignments) {
        const auto& facility = manager_.get_facilities().at(fac_id);

        // 只计算生产设施
        if (facility.type == "贸易站" ||
            facility.type == "制造站" ||
            facility.type == "发电站") {
            total += evaluate_facility(fac_id, operators, eval_state);
        }
    }

    return total;
}

double EfficiencyCalculator::evaluate_facility(
    const std::string& facility_id,
    const std::vector<std::string>& operators,
    const InfrastructureState& state
) {
    double efficiency = 0.0;
    const auto& facility = manager_.get_facilities().at(facility_id);

    for (const auto& op_id : operators) {
        const auto& op = manager_.get_operators().at(op_id);

        for (const auto& skill : op.parsed_skills) {
            // 检查技能是否适用于当前设施
            if (skill.facility != facility.type) continue;

            // 检查是否解锁
            if (op.elite < skill.unlock_elite) continue;
            if (op.elite == skill.unlock_elite && op.level < skill.unlock_level) continue;

            // 计算该技能的所有效果
            for (const auto& effect : skill.effects) {
                switch (effect.type) {
                    case SkillEffectType::FLAT_BONUS:
                        efficiency += effect.value;
                        break;

                    case SkillEffectType::PER_OPERATOR_BONUS:
                        efficiency += calc_per_operator_bonus(effect, operators, state);
                        break;

                    case SkillEffectType::PER_PRODUCTION_LINE:
                        efficiency += calc_per_line_bonus(effect, state);
                        break;

                    case SkillEffectType::SYNERGY_SPECIFIC:
                        efficiency += calc_synergy_specific(effect, operators);
                        break;

                    case SkillEffectType::SYNERGY_GROUP:
                        efficiency += calc_synergy_group(effect, operators);
                        break;

                    case SkillEffectType::VARIABLE_CONSUME:
                        efficiency += calc_variable_consume(effect, state);
                        break;

                    default:
                        break;
                }
            }
        }
    }

    return efficiency;
}
```

```cpp
// 特定干员联动: "与德克萨斯一起工作时+65%"
double EfficiencyCalculator::calc_synergy_specific(
    const SkillEffect& effect,
    const std::vector<std::string>& operators
) {
    // 检查目标干员是否在同设施
    for (const auto& op_id : operators) {
        if (is_operator(op_id, effect.condition_operator)) {
            return effect.value;  // 触发联动
        }
    }
    return 0.0;
}

// 组织联动: "每个[格拉斯哥帮]干员+5%"
double EfficiencyCalculator::calc_per_operator_bonus(
    const SkillEffect& effect,
    const std::vector<std::string>& operators,
    const InfrastructureState& state
) {
    int count = count_group_members(operators, effect.condition_group);
    return effect.value * (count / effect.condition_count);
}

// 变量消费: "每1点人间烟火，生产力+1%"
double EfficiencyCalculator::calc_variable_consume(
    const SkillEffect& effect,
    const InfrastructureState& state
) {
    auto it = state.variables.find(effect.variable_name);
    if (it == state.variables.end()) return 0.0;

    double var_value = it->second;
    return effect.value * var_value;
}
```

### 5.2 模拟退火主循环

```cpp
SchedulePlan ScheduleOptimizer::optimize(const OptimizerConfig& config) {
    auto start_time = std::chrono::steady_clock::now();

    // ===== Step 1: 生成初始解 =====
    InfrastructureState current = greedy_initialize();
    double current_score = calculator_.evaluate(current);

    InfrastructureState best = current;
    double best_score = current_score;

    stats_.initial_score = current_score;

    // ===== Step 2: 模拟退火 =====
    double temperature = config.initial_temperature;
    int plateau_count = 0;

    std::uniform_real_distribution<> uniform(0.0, 1.0);

    for (int iter = 0; iter < config.max_iterations; ++iter) {
        // 检查温度
        if (temperature < config.min_temperature) break;

        // 检查停滞
        if (plateau_count > config.plateau_threshold) {
            // 重新加热
            temperature = config.initial_temperature * 0.5;
            plateau_count = 0;
        }

        // 生成邻域解
        InfrastructureState neighbor = generate_neighbor(current, config);

        // 验证合法性
        if (!is_valid_state(neighbor)) continue;

        // 评估
        double neighbor_score = calculator_.evaluate(neighbor);
        double delta = neighbor_score - current_score;

        // 接受准则
        bool accept = false;
        if (delta > 0) {
            accept = true;  // 更优解，直接接受
        } else {
            // Metropolis 准则
            double prob = std::exp(delta / temperature);
            accept = uniform(rng_) < prob;
        }

        if (accept) {
            current = neighbor;
            current_score = neighbor_score;
            stats_.accepted_moves++;

            if (current_score > best_score) {
                best = current;
                best_score = current_score;
                plateau_count = 0;
            } else {
                plateau_count++;
            }
        } else {
            stats_.rejected_moves++;
            plateau_count++;
        }

        // 降温
        temperature *= config.cooling_rate;
        stats_.total_iterations = iter + 1;
    }

    // ===== Step 3: 记录统计 =====
    auto end_time = std::chrono::steady_clock::now();
    stats_.final_score = best_score;
    stats_.improvement_percent = (best_score - stats_.initial_score) / stats_.initial_score * 100;
    stats_.time_ms = std::chrono::duration<double, std::milli>(end_time - start_time).count();

    return build_plan(best);
}
```

### 5.3 贪心初始解

```cpp
InfrastructureState ScheduleOptimizer::greedy_initialize() {
    InfrastructureState state;
    std::set<std::string> assigned;

    // 按设施优先级排序
    std::vector<std::string> facility_order;
    for (const auto& [fac_id, fac] : manager_.get_facilities()) {
        if (fac.type == "贸易站" || fac.type == "制造站" || fac.type == "发电站") {
            facility_order.push_back(fac_id);
        }
    }

    // 对每个设施，选择效率最高的干员
    for (const auto& fac_id : facility_order) {
        const auto& facility = manager_.get_facilities().at(fac_id);

        // 获取该设施的候选干员及效率
        std::vector<std::pair<std::string, double>> candidates;

        for (const auto& [op_id, op] : manager_.get_operators()) {
            if (assigned.count(op_id)) continue;
            if (!has_skill_for_facility(op, facility.type)) continue;

            double eff = calculate_base_efficiency(op, facility.type);
            candidates.emplace_back(op_id, eff);
        }

        // 按效率排序
        std::sort(candidates.begin(), candidates.end(),
                  [](const auto& a, const auto& b) { return a.second > b.second; });

        // 选择前N个
        for (int i = 0; i < facility.slots && i < candidates.size(); ++i) {
            state.assignments[fac_id].push_back(candidates[i].first);
            assigned.insert(candidates[i].first);
        }
    }

    return state;
}
```

### 5.4 邻域操作

```cpp
InfrastructureState ScheduleOptimizer::generate_neighbor(
    const InfrastructureState& current,
    const OptimizerConfig& config
) {
    std::uniform_real_distribution<> dist(0.0, 1.0);
    double r = dist(rng_);

    double threshold1 = config.swap_weight;
    double threshold2 = threshold1 + config.move_weight;

    if (r < threshold1) {
        return swap_operators(current);
    } else if (r < threshold2) {
        return move_operator(current);
    } else {
        return replace_operator(current);
    }
}

// 交换两个设施中的干员
InfrastructureState ScheduleOptimizer::swap_operators(const InfrastructureState& state) {
    InfrastructureState result = state;

    // 随机选择两个设施
    std::vector<std::string> facilities;
    for (const auto& [fac_id, ops] : state.assignments) {
        if (!ops.empty()) facilities.push_back(fac_id);
    }

    if (facilities.size() < 2) return result;

    std::uniform_int_distribution<> fac_dist(0, facilities.size() - 1);
    int idx1 = fac_dist(rng_);
    int idx2 = fac_dist(rng_);
    while (idx2 == idx1) idx2 = fac_dist(rng_);

    const std::string& fac1 = facilities[idx1];
    const std::string& fac2 = facilities[idx2];

    auto& ops1 = result.assignments[fac1];
    auto& ops2 = result.assignments[fac2];

    // 随机选择各设施中的一个干员
    std::uniform_int_distribution<> op_dist1(0, ops1.size() - 1);
    std::uniform_int_distribution<> op_dist2(0, ops2.size() - 1);

    int op_idx1 = op_dist1(rng_);
    int op_idx2 = op_dist2(rng_);

    // 检查交换后是否合法（干员是否有对应设施技能）
    const auto& facility1 = manager_.get_facilities().at(fac1);
    const auto& facility2 = manager_.get_facilities().at(fac2);
    const auto& op1 = manager_.get_operators().at(ops1[op_idx1]);
    const auto& op2 = manager_.get_operators().at(ops2[op_idx2]);

    if (!has_skill_for_facility(op1, facility2.type)) return state;
    if (!has_skill_for_facility(op2, facility1.type)) return state;

    // 执行交换
    std::swap(ops1[op_idx1], ops2[op_idx2]);

    return result;
}

// 替换干员
InfrastructureState ScheduleOptimizer::replace_operator(const InfrastructureState& state) {
    InfrastructureState result = state;

    // 获取未分配的干员
    auto unassigned = get_unassigned_operators(state);
    if (unassigned.empty()) return result;

    // 随机选择一个设施和其中的干员
    std::vector<std::pair<std::string, int>> assigned_ops;
    for (const auto& [fac_id, ops] : state.assignments) {
        for (int i = 0; i < ops.size(); ++i) {
            assigned_ops.emplace_back(fac_id, i);
        }
    }

    if (assigned_ops.empty()) return result;

    std::uniform_int_distribution<> assigned_dist(0, assigned_ops.size() - 1);
    auto [fac_id, op_idx] = assigned_ops[assigned_dist(rng_)];

    // 从未分配干员中选择一个兼容的
    const auto& facility = manager_.get_facilities().at(fac_id);
    std::vector<std::string> compatible;
    for (const auto& op_id : unassigned) {
        const auto& op = manager_.get_operators().at(op_id);
        if (has_skill_for_facility(op, facility.type)) {
            compatible.push_back(op_id);
        }
    }

    if (compatible.empty()) return result;

    std::uniform_int_distribution<> compat_dist(0, compatible.size() - 1);
    std::string new_op = compatible[compat_dist(rng_)];

    // 执行替换
    result.assignments[fac_id][op_idx] = new_op;

    return result;
}
```
